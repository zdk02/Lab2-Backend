
#drop schema LibraryDB;
create schema LibraryDB;

use LibraryDB;

create table Authors (
	author_id int auto_increment primary key,
    name varchar(20) not null,
    birth_date date not null,
    country varchar(20)
);

create table Books (
	book_id int auto_increment primary key,
    title varchar(30) not null,
    author_id int not null,
    isbn varchar(20) not null unique,
    published_year int not null,
    FOREIGN KEY (author_id) REFERENCES Authors(author_id)
);

create table Borrowers (
	borrower_id int auto_increment primary key,
    name varchar(20) not null,
    email varchar(30) not null unique,
    phone varchar(20)
);

create table Loans (
	loan_id int auto_increment primary key,
	book_id int not null,
    borrower_id int not null,
	loan_date date not null,
    return_date date,
    due_date date not null,
	returned bool,
    FOREIGN KEY (book_id) REFERENCES Books(book_id),
    FOREIGN KEY (borrower_id) REFERENCES Borrowers(borrower_id)
);

create view popular_books as
select b.book_id, b.title, count(l.loan_id) as total_loans
from books b
join loans l on b.book_id = l.book_id
group by b.book_id, b.title
order by total_loans desc;

delimiter $$

create procedure borrow_book(
    in p_book_id int,
    in p_borrower_id int,
    in p_loan_date date
)
begin
    insert into loans (book_id, borrower_id, loan_date, due_date, returned)
    values (p_book_id, p_borrower_id, p_loan_date, date_add(p_loan_date, interval 14 day), false);
end$$

create procedure return_book(
    in p_loan_id int,
    in p_return_date date
)
begin
    update loans
    set returned = true,
        return_date = p_return_date
    where loan_id = p_loan_id;
end$$

delimiter ;
